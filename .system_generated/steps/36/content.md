  (function() {
    // Prevent multiple initializations
    if (window.WhatsAppWidgetInitialized) return;
    window.WhatsAppWidgetInitialized = true;

    const widgetLink = document.getElementById('wa-widget-link');
    const WHATSAPP_PHONE = "201015380800"; // الرقم الخاص بك
    const PREFILLED_MESSAGE = "مرحباً بكم، أود الاستفسار عن خدماتكم المتاحة."; // الرسالة المجهزة للعميل

    if (widgetLink) {
      const isMobile = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
      const textParam = encodeURIComponent(PREFILLED_MESSAGE);
      if (isMobile) {
        // Deep link scheme: فتح التطبيق المباشر (العادي أو البيزنس) بدون متصفح
        widgetLink.href = `whatsapp://send?phone=${WHATSAPP_PHONE}&text=${textParam}`;
        widgetLink.removeAttribute('target'); // إلغاء فتح نافذة جديدة فارغة في متصفح الموبايل
      } else {
        // Universal web link: للكمبيوتر يفتح ويب أو التطبيق
        widgetLink.href = `https://wa.me/${WHATSAPP_PHONE}?text=${textParam}`;
      }
    }
  })();
</script>
